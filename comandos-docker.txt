# Rodar container psql
docker run --name librarydb -e POSTGRES_PASSWORD=postgres -e POSTGRES_USER=postgres -e POSTGRES_DB=library -p 5432:5432 postgres:16.3

#Rodar pgadmin4
docker run --name pgadmin -p 15432:80 -e PGADMIN_DEFAULT_EMAIL=admin@admin.com -e PGADMIN_DEFAULT_PASSWORD=Admin#12345 dpage/pgadmin4

# Comando docker para listar container CMD
docker ps

# Comando para parar o container
docker stop librarydb

# Comando para iniciar o container
docker start librarydb

# Comando para mostrar todos containers
docker ps -a

# Comando para deletar um container
docker container rm librarydb

# Comando para criar network no docker
docker  network create library-network

#Comando para subir container com network
docker run --name librarydb -e POSTGRES_PASSWORD=postgres -e POSTGRES_USER=postgres -e POSTGRES_DB=library -p 5482:5432 --network library-network postgres
docker run --name pgadmin4 -p 15432:80 -e PGADMIN_DEFAULT_EMAIL=admin@admin.com -e PGADMIN_DEFAULT_PASSWORD=admin --network library-network dpage/pgadmin4

# Comando para buildar imagem
docker build --tag kaiocandido/libraryapi

#Acesso ao SWAGGER
http://localhost:54923/swagger-ui/index.html#/

# Comando para buildar java
./mvnw clean package -DskipTests
java -jar .\target\libraryapi-0.0.1-SNAPSHOT.jar

# Subir aplicação no docker
docker run --name libraryapi-production --network library-network -d -p 54923:54923 -p 9090:9090 -e DATASOURCE_URL="jdbc:postgresql://librarydb-prod:5432/library" -e DATASOURCE_USERNAME="postgresprod" -e DATASOURCE_PASSWORD="postgresprod" -e GOOGLE_CLIENT_ID="SEU_CLIENT_ID_AQUI" -e GOOGLE_CLIENT_SECRET="SEU_CLIENT_SECRET_AQUI" kaiocandido/libraryapi

# Subir na AWS
docker run --name libraryapi  -p 54923:54923 -p 9090:9090 -e DATASOURCE_URL="jdbc:postgresql://libraryapi-databse.cn66ywukimua.us-east-2.rds.amazonaws.com:5432/library" -e DATASOURCE_USERNAME="postgres" -e DATASOURCE_PASSWORD="postgres" -e GOOGLE_CLIENT_ID="SEU_CLIENT_ID_AQUI" -e GOOGLE_CLIENT_SECRET="SEU_CLIENT_SECRET_AQUI" kaiocandf/libraryapi
